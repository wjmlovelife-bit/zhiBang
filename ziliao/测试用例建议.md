```ts
it('3. ⚠ 行为记录：跨日时源码 bug —— 不传当日 prefix 时会被昨日 code 污染...', () => {

  setMock(mock, 'crmWorkLog.findFirst', { code: 'RJ202609210099' });
  const code = await service.nextCode();
  expect(code).toMatch(/^RJ\d{8}0100$/); // 行为记录
});
```

```ts
export function normalizeDates(v: any): any {
  if (Array.isArray(v)) return v.map(normalizeDates);
  if (v && typeof v === 'object' && !(v instanceof Date) &&
      !(v.constructor?.name === 'Decimal')) {
    const out: any = {};
    for (const k of Object.keys(v)) out[k] = normalizeDates(v[k]);
    return out;
  }
  if (typeof v === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(v)) {
    return v + 'T00:00:00.000Z';
  }
  return v;
}

export function cleanEmpty(obj: any, fields: string[]): any {
  if (!obj) return obj;
  for (const f of fields) {
    if (obj[f] === '' || obj[f] === null) obj[f] = undefined;
  }
  return obj;
}

export function splitContactName(s: string | null | undefined): string[] {
  if (!s) return [];
  return s.split(';').map(x => x.trim()).filter(Boolean);
}
```

### 8.4 e2e 关键教训（v3 总结）

1. **日期边界**：用 `new Date('YYYY-MM-DD')` 永远是 UTC 00:00，与源码的"半开区间"易冲突。建议明确写 `new Date('YYYY-MM-DDT08:00:00Z')` 表示当日工作时间
2. **类型统一**：upsert 与 getByWeek 入参类型必须一致（string vs Date 是 v2 F-3 根因）
3. **类名拼写**：import 必须严格匹配源码导出的大小写（**`WorkLogService` 大写 L**）——这是 v2 F-4 的真实根因，**不是框架误报**
4. **TZ 依赖**：跨月跨年测试要明确 TZ（happy-dom 默认 UTC，源码可能用 Asia/Shanghai）

### 8.5 命名规范

- **后端 spec**：`{module}.{aspect}.spec.ts`（如 `customer-edge.spec.ts`、`sys.apply-role.spec.ts`）
- **前端 spec**：`{源文件路径}.spec.ts`（如 `utils/payload.spec.ts`、`store/user.spec.ts`）
- **测试 describe**：源类名或函数名
- **测试 it**：中文 `should + 动词 + 对象 + 期望`，**⚠ 行为记录：源码 bug ...** 标记已知问题

### 8.6 禁止做法

- ❌ 不要 `expect(true).toBe(true)` 恒真断言
- ❌ 不要把 `assert(... || ..., ...)` 写成"必然通过"
- ❌ **不要 inline 副本测源函数（必须 import）** —— v3 B-14 / L-1 加强版教训
